#!/bin/sh
# LXQt Safe Switch
# POSIX sh | Detecta a distribuição, instala LXQt e configura o SDDM.
# Não remove pacotes, não reinicia e não interrompe a sessão gráfica atual.
# A alteração do display manager é preparada para o próximo início do sistema.

set -u
umask 077
export LC_ALL=C
export DEBIAN_FRONTEND=noninteractive

PROGRAM="LXQt Safe Switch"
LOCK="/run/lock/lxqt-safe-switch.lock"
BACKUP_DIR="/var/backups/lxqt-safe-switch"
LOG="/var/log/lxqt-safe-switch.log"
TMP=""
LOCKED=0
CHANGED_DM=0
OLD_DM=""
OLD_DM_ENABLED=""
OLD_DM_TARGET=""
NEW_DM="sddm"
DISTRO_ID=""
DISTRO_LIKE=""
PKG_MANAGER=""
INSTALL_OK=0

say() {
    printf '%s\n' "$*"
}

info() {
    say "[INFO] $*"
}

warn() {
    say "[AVISO] $*" >&2
}

error() {
    say "[ERRO] $*" >&2
}

pause() {
    printf '\nPressione ENTER para continuar...'
    IFS= read -r unused || :
}

cleanup() {
    if [ -n "${TMP:-}" ] && [ -d "$TMP" ]; then
        rm -rf -- "$TMP"
    fi

    if [ "$LOCKED" -eq 1 ]; then
        rmdir "$LOCK" 2>/dev/null || :
    fi
}

rollback_dm() {
    [ "$CHANGED_DM" -eq 1 ] || return 0

    warn "Tentando restaurar o display manager anterior."

    systemctl disable "$NEW_DM.service" >/dev/null 2>&1 || :

    if [ -n "$OLD_DM" ]; then
        systemctl enable "$OLD_DM.service" >/dev/null 2>&1 || :
    fi

    if [ -n "$OLD_DM_TARGET" ] && [ -e "$BACKUP_DIR/display-manager.service" ]; then
        rm -f /etc/systemd/system/display-manager.service
        ln -s "$OLD_DM_TARGET" /etc/systemd/system/display-manager.service \
            2>/dev/null || :
    fi

    systemctl daemon-reload >/dev/null 2>&1 || :
    CHANGED_DM=0
}

on_signal() {
    error "Execução interrompida."
    if [ "$CHANGED_DM" -eq 1 ]; then
        rollback_dm
    fi
    exit 130
}

trap cleanup EXIT
trap on_signal HUP INT TERM

require_root() {
    if [ "$(id -u)" -ne 0 ]; then
        error "Execute como administrador:"
        say "sudo sh $0"
        exit 1
    fi
}

detect_distro() {
    if [ ! -r /etc/os-release ]; then
        error "Não foi possível identificar a distribuição."
        exit 1
    fi

    DISTRO_ID=$(
        . /etc/os-release
        printf '%s' "${ID:-unknown}"
    )

    DISTRO_LIKE=$(
        . /etc/os-release
        printf '%s' "${ID_LIKE:-}"
    )

    if command -v apt-get >/dev/null 2>&1; then
        PKG_MANAGER="apt"
    elif command -v dnf >/dev/null 2>&1; then
        PKG_MANAGER="dnf"
    elif command -v pacman >/dev/null 2>&1; then
        PKG_MANAGER="pacman"
    elif command -v zypper >/dev/null 2>&1; then
        PKG_MANAGER="zypper"
    else
        error "Gerenciador de pacotes não suportado."
        exit 1
    fi

    info "Distribuição: $DISTRO_ID"
    info "Família declarada: ${DISTRO_LIKE:-não informada}"
    info "Gerenciador de pacotes: $PKG_MANAGER"
}

check_environment() {
    if [ -n "${SSH_CONNECTION:-}" ] || [ -n "${SSH_TTY:-}" ]; then
        error "Execução remota via SSH bloqueada por segurança."
        error "Execute localmente no computador."
        exit 1
    fi

    if [ ! -d /run/systemd/system ] || ! command -v systemctl >/dev/null 2>&1; then
        error "Este script exige systemd para configurar o display manager."
        error "Nenhuma alteração foi realizada."
        exit 1
    fi

    if ! systemctl show-environment >/dev/null 2>&1; then
        error "O gerenciador systemd não está acessível."
        exit 1
    fi

    if [ -d "$LOCK" ]; then
        error "Já existe uma execução em andamento ou um bloqueio anterior."
        error "Verifique manualmente: $LOCK"
        exit 1
    fi

    if ! mkdir "$LOCK" 2>/dev/null; then
        error "Não foi possível criar o bloqueio de execução."
        exit 1
    fi
    LOCKED=1

    if command -v pgrep >/dev/null 2>&1; then
        if pgrep -x apt >/dev/null 2>&1 ||
           pgrep -x apt-get >/dev/null 2>&1 ||
           pgrep -x dpkg >/dev/null 2>&1 ||
           pgrep -x dnf >/dev/null 2>&1 ||
           pgrep -x rpm >/dev/null 2>&1 ||
           pgrep -x pacman >/dev/null 2>&1 ||
           pgrep -x zypper >/dev/null 2>&1; then
            error "Outro gerenciador de pacotes está em execução."
            exit 1
        fi
    fi

    if [ "$(df -Pk / | awk 'NR==2 {print $4}')" -lt 3145728 ]; then
        error "Espaço livre insuficiente na partição raiz."
        error "São necessários pelo menos 3 GiB livres."
        exit 1
    fi

    TMP=$(mktemp -d /tmp/lxqt-safe-switch.XXXXXX) || {
        error "Não foi possível criar diretório temporário."
        exit 1
    }

    if ! mkdir -p "$BACKUP_DIR"; then
        error "Não foi possível criar o diretório de backup."
        exit 1
    fi

    chmod 700 "$BACKUP_DIR" 2>/dev/null || :
    touch "$LOG" || {
        error "Não foi possível criar o arquivo de log."
        exit 1
    }
    chmod 600 "$LOG" 2>/dev/null || :

    info "Verificações básicas concluídas."
}

detect_current_dm() {
    OLD_DM=""
    OLD_DM_ENABLED=""
    OLD_DM_TARGET=""

    for dm in sddm lightdm gdm3 gdm lxdm ly; do
        if systemctl is-active --quiet "$dm.service" 2>/dev/null; then
            OLD_DM="$dm"
            break
        fi
    done

    if [ -z "$OLD_DM" ] && [ -L /etc/systemd/system/display-manager.service ]; then
        OLD_DM_TARGET=$(readlink /etc/systemd/system/display-manager.service)
        resolved=$(readlink -f /etc/systemd/system/display-manager.service 2>/dev/null || :)
        if [ -n "$resolved" ]; then
            OLD_DM=$(basename "$resolved" .service)
        fi
    fi

    if [ -n "$OLD_DM" ]; then
        OLD_DM_ENABLED=$(systemctl is-enabled "$OLD_DM.service" 2>/dev/null || :)
        info "Display manager detectado: $OLD_DM"
        info "Estado de inicialização: ${OLD_DM_ENABLED:-desconhecido}"
    else
        warn "Não foi possível identificar o display manager atual."
        warn "A configuração atual será preservada se a detecção continuar inconclusiva."
    fi

    if [ -L /etc/systemd/system/display-manager.service ]; then
        OLD_DM_TARGET=$(readlink /etc/systemd/system/display-manager.service)
    fi
}

backup_dm_configuration() {
    stamp=$(date '+%Y%m%d-%H%M%S')
    dest="$BACKUP_DIR/$stamp"

    if ! mkdir -p "$dest"; then
        error "Não foi possível preparar o backup."
        return 1
    fi

    {
        printf 'Distribuição: %s\n' "$DISTRO_ID"
        printf 'Gerenciador de pacotes: %s\n' "$PKG_MANAGER"
        printf 'Display manager anterior: %s\n' "${OLD_DM:-desconhecido}"
        printf 'Estado anterior: %s\n' "${OLD_DM_ENABLED:-desconhecido}"
        printf 'Destino anterior do link: %s\n' "${OLD_DM_TARGET:-nenhum}"
        printf 'Data: %s\n' "$stamp"
    } > "$dest/estado.txt"

    if [ -L /etc/systemd/system/display-manager.service ]; then
        cp -P /etc/systemd/system/display-manager.service \
            "$dest/display-manager.service" || return 1
    fi

    if [ -f /etc/X11/default-display-manager ]; then
        cp -p /etc/X11/default-display-manager \
            "$dest/default-display-manager" || return 1
    fi

    info "Backup da configuração salvo em: $dest"
    return 0
}

check_apt_packages() {
    info "Atualizando índices de pacotes APT..."
    apt-get update || {
        error "Falha ao atualizar os índices APT."
        return 1
    }

    info "Verificando disponibilidade de LXQt e SDDM..."

    apt-cache show lxqt >/dev/null 2>&1 || {
        error "O pacote lxqt não está disponível nos repositórios configurados."
        return 1
    }

    apt-cache show sddm >/dev/null 2>&1 || {
        error "O pacote sddm não está disponível nos repositórios configurados."
        return 1
    }

    apt-get -s install lxqt sddm > "$TMP/plan.txt" 2>&1 || {
        cat "$TMP/plan.txt"
        error "A simulação de instalação APT falhou."
        return 1
    }

    if grep -E '^(Remv|Purg) ' "$TMP/plan.txt" >/dev/null 2>&1; then
        cat "$TMP/plan.txt"
        error "A simulação prevê remoção de pacotes. Operação cancelada."
        return 1
    fi

    info "Simulação APT aprovada: nenhuma remoção prevista."
}

check_dnf_packages() {
    for pkg in lxqt-session lxqt-panel pcmanfm-qt qterminal sddm; do
        if ! dnf list --available "$pkg" >/dev/null 2>&1 &&
           ! rpm -q "$pkg" >/dev/null 2>&1; then
            error "Pacote indisponível nos repositórios: $pkg"
            return 1
        fi
    done

    dnf install --assumeno lxqt-session lxqt-panel pcmanfm-qt qterminal sddm \
        > "$TMP/plan.txt" 2>&1 || :

    cat "$TMP/plan.txt"

    if grep -Ei '(^|[[:space:]])(Removing|Erase|Removendo)[[:space:]]|will be removed|removing:' \
        "$TMP/plan.txt" >/dev/null 2>&1; then
        error "A simulação DNF indica remoção de pacotes."
        return 1
    fi

    if ! grep -Ei 'Install|Installing|Nothing to do|already installed|já instalado' \
        "$TMP/plan.txt" >/dev/null 2>&1; then
        error "Não foi possível validar a simulação DNF."
        return 1
    fi

    info "Pré-verificação DNF concluída."
}

check_pacman_packages() {
    for pkg in lxqt sddm; do
        if ! pacman -Si "$pkg" >/dev/null 2>&1 &&
           ! pacman -Q "$pkg" >/dev/null 2>&1; then
            error "Pacote indisponível nos repositórios: $pkg"
            return 1
        fi
    done

    if ! pacman -Sp --needed lxqt sddm > "$TMP/plan.txt" 2>&1; then
        cat "$TMP/plan.txt"
        error "Não foi possível resolver as dependências no Pacman."
        return 1
    fi

    cat "$TMP/plan.txt"
    warn "O Pacman poderá solicitar confirmação adicional caso detecte conflitos."
    warn "Não confirme nenhuma operação que proponha remover pacotes."
}

check_zypper_packages() {
    for pkg in lxqt-session lxqt-panel pcmanfm-qt qterminal sddm; do
        if ! zypper --non-interactive search --match-exact "$pkg" \
            > "$TMP/search.txt" 2>&1; then
            cat "$TMP/search.txt"
            error "Falha ao consultar os repositórios para: $pkg"
            return 1
        fi

        if ! grep -F "$pkg" "$TMP/search.txt" >/dev/null 2>&1 &&
           ! rpm -q "$pkg" >/dev/null 2>&1; then
            error "Pacote não localizado: $pkg"
            return 1
        fi
    done

    zypper --non-interactive install --dry-run \
        lxqt-session lxqt-panel pcmanfm-qt qterminal sddm \
        > "$TMP/plan.txt" 2>&1 || {
        cat "$TMP/plan.txt"
        error "A simulação Zypper falhou."
        return 1
    }

    cat "$TMP/plan.txt"

    if grep -Ei 'Following .* will be REMOVED|The following packages are going to be removed|will be deinstalled' \
        "$TMP/plan.txt" >/dev/null 2>&1; then
        error "A simulação Zypper prevê remoção de pacotes."
        return 1
    fi

    info "Simulação Zypper concluída."
}

preflight_packages() {
    case "$PKG_MANAGER" in
        apt)    check_apt_packages ;;
        dnf)    check_dnf_packages ;;
        pacman) check_pacman_packages ;;
        zypper) check_zypper_packages ;;
        *)
            error "Gerenciador de pacotes não suportado."
            return 1
            ;;
    esac
}

install_lxqt() {
    info "Instalando LXQt e SDDM..."
    info "Nenhum comando de remoção será executado pelo script."

    case "$PKG_MANAGER" in
        apt)
            apt-get install lxqt sddm
            ;;
        dnf)
            dnf install lxqt-session lxqt-panel pcmanfm-qt qterminal sddm
            ;;
        pacman)
            pacman -S --needed lxqt sddm
            ;;
        zypper)
            zypper install lxqt-session lxqt-panel pcmanfm-qt qterminal sddm
            ;;
        *)
            error "Gerenciador de pacotes não suportado."
            return 1
            ;;
    esac
}

validate_lxqt() {
    if [ ! -d /usr/share/xsessions ] && [ ! -d /usr/share/wayland-sessions ]; then
        error "Diretórios de sessões gráficas não encontrados."
        return 1
    fi

    if ! find /usr/share/xsessions /usr/share/wayland-sessions \
        -maxdepth 1 -type f -iname '*lxqt*.desktop' -print 2>/dev/null |
        grep . >/dev/null 2>&1; then
        error "Nenhuma sessão LXQt foi encontrada após a instalação."
        return 1
    fi

    if ! systemctl cat "$NEW_DM.service" >/dev/null 2>&1; then
        error "A unidade systemd de $NEW_DM não está disponível."
        return 1
    fi

    info "Sessão LXQt e unidade SDDM verificadas."
    return 0
}

configure_display_manager() {
    if [ "$OLD_DM" = "$NEW_DM" ]; then
        info "SDDM já é o display manager detectado."
        systemctl enable "$NEW_DM.service" || {
            error "Não foi possível habilitar o SDDM."
            return 1
        }
        return 0
    fi

    if [ -z "$OLD_DM" ]; then
        error "O display manager atual não foi identificado com segurança."
        error "LXQt foi instalado, mas a configuração do display manager não será alterada."
        return 1
    fi

    backup_dm_configuration || {
        error "Não foi possível salvar o backup da configuração."
        return 1
    }

    info "Configurando SDDM para o próximo início do sistema."
    info "A sessão gráfica atual não será encerrada."

    CHANGED_DM=1

    if ! systemctl disable "$OLD_DM.service"; then
        error "Não foi possível desabilitar a inicialização automática de $OLD_DM."
        rollback_dm
        return 1
    fi

    if ! systemctl enable "$NEW_DM.service"; then
        error "Não foi possível habilitar SDDM."
        rollback_dm
        return 1
    fi

    systemctl daemon-reload || {
        error "Falha ao recarregar a configuração do systemd."
        rollback_dm
        return 1
    }

    if ! systemctl is-enabled "$NEW_DM.service" >/dev/null 2>&1; then
        error "SDDM não aparece habilitado após a configuração."
        rollback_dm
        return 1
    fi

    CHANGED_DM=0
    info "SDDM configurado para a próxima inicialização."
    info "O display manager anterior permanece instalado."
    return 0
}

show_menu() {
    clear 2>/dev/null || :

    say "=========================================="
    say "          LXQt SAFE SWITCH"
    say "=========================================="
    say
    say "1) Trocar ambiente para LXQt"
    say "0) Sair"
    say
    printf "Escolha: "
}

main() {
    require_root
    detect_distro

    while :; do
        show_menu
        IFS= read -r choice || exit 0

        case "$choice" in
            1)
                check_environment || {
                    pause
                    continue
                }

                detect_current_dm

                if ! preflight_packages; then
                    error "Pré-verificação falhou. Nenhuma instalação iniciada."
                    pause
                    continue
                fi

                say
                say "A operação instalará o LXQt e o SDDM."
                say "O display manager atual não será removido."
                say "A configuração de inicialização poderá ser alterada."
                say "Não haverá reinicialização automática."
                say
                printf 'Digite INSTALAR para continuar: '
                IFS= read -r confirmation || confirmation=""

                if [ "$confirmation" != "INSTALAR" ]; then
                    info "Operação cancelada pelo usuário."
                    pause
                    continue
                fi

                if ! install_lxqt; then
                    error "A instalação falhou ou foi cancelada."
                    pause
                    continue
                fi

                INSTALL_OK=1

                if ! validate_lxqt; then
                    error "Instalação concluída, mas a validação falhou."
                    error "A configuração do display manager foi preservada."
                    pause
                    continue
                fi

                if ! configure_display_manager; then
                    error "LXQt está instalado, mas o display manager não foi alterado com sucesso."
                    pause
                    continue
                fi

                say
                say "=========================================="
                say "           OPERAÇÃO CONCLUÍDA"
                say "=========================================="
                say "LXQt instalado e SDDM configurado."
                say "O display manager anterior não foi desinstalado."
                say "Nenhuma reinicialização foi executada."
                say "Reinicie manualmente quando desejar aplicar a mudança."
                say
                pause
                ;;

            0)
                exit 0
                ;;

            *)
                warn "Opção inválida."
                pause
                ;;
        esac
    done
}

main "$@"
